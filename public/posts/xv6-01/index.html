1. xv6简介#

xv6 以 Dennis Ritchie 和 Ken Thompson 的 Unix version 6 (v6) 为蓝本 。xv6 大致遵循 Unix v6 的结构和风格，但它是使用ANSI C语言为多核RISC-V 架构实现的。

MIT的操作系统课程6.1810就是以xv6为例，具体阐述操作系统概念。它不仅提供了 Ken Thompson 和 Dennis Ritchie 的 Unix 操作系统所引入的基本接口，还模仿了 Unix 的内部设计。Unix 提供了一个功能较为单一但机制组合良好的接口，从而展现出令人惊讶的通用性。该接口非常成功，以至于现代操作系统——BSD、Linux、macOS、Solaris，甚至在某种程度上也包括 Microsoft Windows——都采用了类似 Unix 的接口。理解 xv6 是理解这些系统以及许多其他系统的良好开端。

xv6提供了若干功能模块：

  • 进程管理
  • 内存管理（物理内存和虚拟内存）
  • I/O和文件描述符
  • 文件系统
  • IPC机制（管道）

xv6与现实世界的操作系统有些不同：

Unix 将“标准的”文件描述符、管道以及便捷的 shell 语法相结合，用于对它们进行操作，这在编写通用可重用程序方面是一项重大进步。这一理念催生了 “软件工具”文化 ，而这正是 Unix 强大功能和流行度的重要原因。Shell 则是第一个所谓的“脚本语言”。Unix 系统调用接口至今仍然在 BSD、Linux 和 macOS 等系统中沿用。

Unix 系统调用接口已通过可移植操作系统接口 (POSIX) 标准进行了标准化。xv6不符合 POSIX 标准：它缺少许多系统调用（包括一些基本调用lseek），而且它提供的许多系统调用也与标准有所不同。我们设计 xv6 的主要目标是在提供简单易用的类 Unix 系统调用接口的同时，保持简洁性和清晰度。一些人已经扩展了 xv6，添加了一些系统调用和一个简单的 C 库，以便运行基本的 Unix 程序。然而，现代内核提供的系统调用和内核服务种类远超 xv6。例如，它们支持网络、窗口系统、用户级线程、多种设备的驱动程序等等。现代内核不断发展且日新月异，并提供了许多超出 POSIX 标准的功能。

Unix 使用一套文件名和文件描述符接口统一访问多种类型的资源（文件、目录和设备）。 这种理念可以扩展到更多类型的资源；一个很好的例子是 Plan 9 ，它将“资源即文件”的概念应用于网络、图形等领域。然而，大多数基于 Unix 的操作系统并没有采用这种做法。

文件系统和文件描述符一直是强大的抽象概念。 即便如此，操作系统接口还有其他模型。Unix 的前身 Multics 将文件存储抽象化，使其看起来像内存，从而产生了一种截然不同的接口风格。Multics 设计的复杂性直接影响了 Unix 的设计者，他们的目标是构建一个更简洁的系统。

4. xv6的Boot过程#

我们一般会通过QEMU（全系统模拟器）启动xv6，QEMU命令 qemu -kernel 会将kernel加载到地址 0x80000000 处，而此处正好是entry.S第一条指令的地址。

关键问题：为什么entry.S会被放到kernel的起始处？

因为链接的过程是由链接脚本 kernel.ld 控制的，在其中指定了将符号_entry放置到kernel起始处。

OUTPUT_ARCH( "riscv" )
ENTRY( _entry )

SECTIONS
{
  /*
   * ensure that entry.S / _entry is at 0x80000000,
   * where qemu's -kernel jumps.
   */
  . = 0x80000000;
  
  .text : {
    kernel/entry.o(_entry)
    *(.text .text.*)
    . = ALIGN(0x1000);
    _trampoline = .;
    *(trampsec)
    . = ALIGN(0x1000);
    ASSERT(. - _trampoline == 0x1000, "error: trampoline larger than one page");
    PROVIDE(etext = .);
  }

  .rodata : {
    . = ALIGN(16);
    *(.srodata .srodata.*) /* do not need to distinguish this from .rodata */
    . = ALIGN(16);
    *(.rodata .rodata.*)
  }

  .data : {
    . = ALIGN(16);
    *(.sdata .sdata.*) /* do not need to distinguish this from .data */
    . = ALIGN(16);
    *(.data .data.*)
  }

  .bss : {
    . = ALIGN(16);
    *(.sbss .sbss.*) /* do not need to distinguish this from .bss */
    . = ALIGN(16);
    *(.bss .bss.*)
  }

  PROVIDE(end = .);
}

xv6是一个支持多核心的操作系统，因此下面整个启动过程在所有核心上均会执行。

xv6的启动过程为：entry.S->start.c-> main.c。

首先来看entry.S，其中为每个核心的执行设置了内核栈:

        # qemu -kernel loads the kernel at 0x80000000
        # and causes each hart (i.e. CPU) to jump there.
        # kernel.ld causes the following code to
        # be placed at 0x80000000.
.section .text
.global _entry
_entry:
        # set up a stack for C.
        # stack0 is declared in start.c,
        # with a 4096-byte stack per CPU.
        # sp = stack0 + ((hartid + 1) * 4096)
        la sp, stack0
        li a0, 1024*4       # 4K
        csrr a1, mhartid    # 将hartid存入a1中
        addi a1, a1, 1
        mul a0, a0, a1      # 得到hart对应内核栈的栈顶地址
        add sp, sp, a0
        # jump to start() in start.c
        call start
spin:
        j spin

在_entry的末尾跳转到start函数（kernel/start.c）：

// entry.S needs one stack per CPU.
__attribute__((aligned(16))) char stack0[4096 * NCPU];

// entry.S jumps here in machine mode on stack0.
void
start()
{
  // set M Previous Privilege mode to Supervisor, for mret.
  unsigned long x = r_mstatus();
  x &= ~MSTATUS_MPP_MASK;
  x |= MSTATUS_MPP_S;
  w_mstatus(x);

  // set M Exception Program Counter to main, for mret.
  // requires gcc -mcmodel=medany
  w_mepc((uint64)main);
	
  // disable paging for now.
  w_satp(0);

  // delegate all interrupts and exceptions to supervisor mode.
  w_medeleg(0xffff);
  w_mideleg(0xffff);
  w_sie(r_sie() | SIE_SEIE | SIE_STIE);

  // configure Physical Memory Protection to give supervisor mode
  // access to all of physical memory.
  w_pmpaddr0(0x3fffffffffffffull);
  w_pmpcfg0(0xf);

  // enable hardware updates of page table A and D bits
  w_menvcfg(r_menvcfg() | MENVCFG_ADUE);

  // ask for clock interrupts.
  timerinit();
	
  // keep each CPU's hartid in its tp register, for cpuid().
  int id = r_mhartid();
  w_tp(id);

  // switch to supervisor mode and jump to main().
  asm volatile("mret");
}

其中做了一些初始化工作，包括从机器模式（Machine mode）到监管者模式（Supervisor Mode）的切换、初始化timer等。最后通过mret命令条赚到main函数（kernel/main.c）：

#include "types.h"
#include "param.h"
#include "memlayout.h"
#include "riscv.h"
#include "defs.h"

volatile static int started = 0;

// start() jumps here in supervisor mode on all CPUs.
void
main()
{
  if (cpuid() == 0) {
    consoleinit();
    printkinit();
    printk("\n");
    printk("xv6 kernel is booting\n");
    printk("\n");
    kinit();            // physical page allocator
    kvminit();          // create kernel page table
    kvminithart();      // turn on paging
    procinit();         // process table
    trapinit();         // trap vectors
    trapinithart();     // install kernel trap vector
    plicinit();         // set up interrupt controller
    plicinithart();     // ask PLIC for device interrupts
    binit();            // buffer cache
    iinit();            // inode table
    fileinit();         // file table
    virtio_disk_init(); // emulated hard disk
    userinit();         // first user process
		
    __atomic_store_n(&started, 1, __ATOMIC_RELEASE);
  } else {
    while (__atomic_load_n(&started, __ATOMIC_ACQUIRE) == 0)
      ;

    printk("hart %d starting\n", cpuid());
    kvminithart();  // turn on paging
    trapinithart(); // install kernel trap vector
    plicinithart(); // ask PLIC for device interrupts
  }

  scheduler();
}

在main函数中对内核各模块进行了初始化。注意：只有第一个执行到main函数的处理器核心（以后就称其为hart）才会执行这段初始化操作，并进入userinit()函数，启动第一个用户程序。其余的hart只会进行少数几个初始化步骤，并等待调度器的调度。

最后来看看第一个用户程序是如何启动，在userinit()中会构造第一个用户进程：

// Set up first user process.
void
userinit(void)
{
  struct proc *p;

  p = allocproc();
  initproc = p;

  p->cwd = namei("/");

  p->state = RUNNABLE;

  release(&p->lock);
}

其中allocproc()会分配进程相关的资源，其中会将进程上下文中的ra字段设置为forkret：

p->context.ra = (uint64)forkret;

最后返回到main()函数，执行scheduler()函数。为了避免深入复杂的调度过程，这里跳过scheduler()函数的执行，只需要知道通过scheduler()函数，最终会进入forkret函数（kernel/proc.c）：

// A fork child's very first scheduling by scheduler()
// will swtch to forkret.
void
forkret(void)
{
  extern char userret[];
  static int first = 1;
  struct proc *p = myproc();

  // Still holding p->lock from scheduler.
  release(&p->lock);
	
  if (__atomic_load_n(&first, __ATOMIC_ACQUIRE)) {
    // File system initialization must be run in the context of a
    // regular process (e.g., because it calls sleep), and thus cannot
    // be run from main().
    fsinit(ROOTDEV);

    // ensure other cores see first=0.
    __atomic_store_n(&first, 0, __ATOMIC_RELEASE);
		
    // We can invoke kexec() now that file system is initialized.
    // Put the return value (argc) of kexec into a0.
    p->trapframe->a0 = kexec("/init", (char *[]){"/init", 0});
    if (p->trapframe->a0 == -1) {
      panic("exec");
    }
  }

  // return to user space, mimicing usertrap()'s return.
  prepare_return();
  uint64 satp = MAKE_SATP(p->pagetable);
  uint64 trampoline_userret = TRAMPOLINE + (userret - trampoline);
  ((void (*)(uint64))trampoline_userret)(satp);
}

其中的first是一个静态变量，初始值为1。在 if 语句中会用原子操作检查 first 是否为1：

  • 如果是，说明当前hart是第一次执行到此处的hart，即它将启动的是第一个用户程序
    • 首先调用fsinit()初始化文件系统。
    • 用原子操作将first的值置为0。
    • 然后执行程序 init
  • 否则，说明该hart执行的是其他用户程序，跳过上述步骤

这里的 init 来自user/init.c：

// init: The initial user-level program

#include "kernel/types.h"
#include "kernel/stat.h"
#include "kernel/spinlock.h"
#include "kernel/sleeplock.h"
#include "kernel/fs.h"
#include "kernel/file.h"
#include "user/user.h"
#include "kernel/fcntl.h"

char *argv[] = {"sh", 0};

int
main(void)
{
  int pid, wpid;

  if (open("console", O_RDWR) < 0) {
    mknod("console", CONSOLE, 0);
    open("console", O_RDWR);
  }
  dup(0); // stdout
  dup(0); // stderr
	
  for (;;) {
    printf("init: starting sh\n");
    pid = fork();
    if (pid < 0) {
      printf("init: fork failed\n");
      exit(1);
    }
    if (pid == 0) {
      exec("sh", argv);
      printf("init: exec sh failed\n");
      exit(1);
    }

    for (;;) {
      // this call to wait() returns if the shell exits,
      // or if a parentless process exits.
      wpid = wait((int *)0);
      if (wpid == pid) {
        // the shell exited; restart it.
        break;
      } else if (wpid < 0) {
        printf("init: wait returned an error\n");
        exit(1);
      } else {
        // it was a parentless process; do nothing.
      }
    }
  }
}

首先打开了 0, 1, 2 consle文件，作为标准输入、标准输出和标准错误。fork出一个子进程，并通过exec启动shell（user/sh.c），然后进入死循环等待shell结束。

5. 添加用户程序的方法#

6. xv6的系统调用#

从形式上来说，系统调用的使用类似函数调用，但两者有着本质上的不同。

  • 函数调用本质上只是将当前函数的上下文保存在栈中，然后跳转到同一地址空间下的另一个函数去执行（改变PC的值），执行完毕后再将保存在栈中的返回地址重新载入PC，继续原函数的执行。整个过程都在用户态进行。
  • 系统调用则不同，系统调用的目的是获取内核提供的服务，所以必须从用户态进入到内核态，这是其与函数调用的本质区别。

关键问题：如何从用户态进入内核态？

不同体系结构有着不同的切换方式，x86通过著名的0x80中断实现（int 0x80），其本质是内核在初始化IDT（中断描述符表）时，将0x80中断的权限设置为可被用户态访问（DLP=3）。RISC-V则提供了专门的指令ecall，下面介绍一些该指令：

ecall 是 RISC-V 中用于主动触发环境调用异常的指令，是用户态程序请求操作系统内核服务（即系统调用）的核心机制。它本身并不执行具体的服务，而是像一个“门铃”，通知更高权限的执行环境（如操作系统内核）来处理请求。

你执行一条 ecall 指令时，硬件会执行如下动作：

  1. 记录返回地址：将 ecall 指令自身的地址（而不是下一条指令的地址）保存到接收异常的权限级的 epc 寄存器中（例如 U-mode 触发、S-mode 处理，则存入 sepc）。
  2. 设置异常原因：在对应的 cause 寄存器（如 scause）中记录异常类型，以便处理程序知道这是来自哪个权限级的 ecall。
  3. 跳转：将 PC 设置为该权限级预设的陷阱向量地址（如 S-mode 的 stvec），开始执行内核的入口代码。

在内核启动应用程序，从内核态返回用户态时已经将stvec设置为切换指令的地址，这部分内容在forkret() → prepare_return()：

prepare_return(void)
{
  ......

  // send syscalls, interrupts, and exceptions to uservec in trampoline.S
  uint64 trampoline_uservec = TRAMPOLINE + (uservec - trampoline);
  w_stvec(trampoline_uservec);
  
  ......
}

        sfence.vma zero, zero

        jalr t0

因为ecall指令仅记录了少量信息（返回地址和trap原因），在uservec中需要保存完整的应用程序在用户态的执行上下文，并切换到内核上下文，然后调用usertrap()函数。

//
// handle an interrupt, exception, or system call from user space.
// called from, and returns to, trampoline.S
// return value is user satp for trampoline.S to switch to.
//
uint64
usertrap(void)
{
  int which_dev = 0;

  if ((r_sstatus() & SSTATUS_SPP) != 0)
    panic("usertrap: not from user mode");

  // send interrupts and exceptions to kerneltrap(),
  // since we're now in the kernel.
  w_stvec((uint64)kernelvec); //DOC: kernelvec

  struct proc *p = myproc();

  // save user program counter.
  p->trapframe->epc = r_sepc();

  if (r_scause() == 8) {
    // system call

    if (killed(p))
      kexit(-1);

    // sepc points to the ecall instruction,
    // but we want to return to the next instruction.
    p->trapframe->epc += 4;

    // an interrupt will change sepc, scause, and sstatus,
    // so enable only now that we're done with those registers.
    intr_on();

    syscall();
  } else if ((which_dev = devintr()) != 0) {
    // ok
  } else if ((r_scause() == 15 || r_scause() == 13) &&
             vmfault(p->pagetable, p->sz, r_stval(),
                     (r_scause() == 13) ? 1 : 0) != 0) {
    // page fault on lazily-allocated page
  } else {
    printk("usertrap(): unexpected scause 0x%lx pid=%d\n", r_scause(), p->pid);
    printk("            sepc=0x%lx stval=0x%lx\n", r_sepc(), r_stval());
    setkilled(p);
  }

  if (killed(p))
    kexit(-1);

  // give up the CPU if this is a timer interrupt.
  if (which_dev == 2)
    yield();

  prepare_return();

  // the user page table to switch to, for trampoline.S
  uint64 satp = MAKE_SATP(p->pagetable);

  // return to trampoline.S; satp value in a0.
  return satp;
}

在usertrap()函数中，会根据scause的值（trap原因）执行不同的处理。在xv6中，系统调用时会将scause置为8，因此会去调用syscall()函数：

void
syscall(void)
{
  int num;
  struct proc *p = myproc();

  num = p->trapframe->a7;
  if (num > 0 && num < NELEM(syscalls) && syscalls[num]) {
    // Use num to lookup the system call function for num, call it,
    // and store its return value in p->trapframe->a0
    p->trapframe->a0 = syscalls[num]();
  } else {
    printk("%d %s: unknown sys call %d\n", p->pid, p->name, num);
    p->trapframe->a0 = -1;
  }
}
